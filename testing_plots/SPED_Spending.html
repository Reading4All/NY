// Guard function matching DistrictELAGrowth.qmd toolbar architecture
function guard(fn, options = {}, countyView) {
  const {
    submitLabel = "Submit",
    resetLabel = "Reset",
    required = false,
    resubmit = true,
    width = "100%",
    justify = "start"
  } = options;
  
  const onSubmit = () => {
    value = input.value;
    submit.disabled = !resubmit || invalid;
    reset.disabled = true;
    wrapper.dispatchEvent(new Event("input", { bubbles: true }));
  };
  const onReset = () => {
    input.value = value;
    submit.disabled = !resubmit || invalid;
    reset.disabled = true;
  };
  const onInnerInputCapture = () => {
    invalid = true;
    submit.disabled = true;
  };
  const onInnerInput = e => {
    e.stopPropagation();
    submit.disabled = false;
    reset.disabled = false;
  };
  const submit = htl.html`<button ${ { disabled: !resubmit && !required, onclick: onSubmit } } class="btn btn-primary btn-sm">${submitLabel}</button>`;
  const reset = htl.html`<button ${ { disabled: true, onclick: onReset } } class=\"btn btn-secondary btn-sm\">${resetLabel}</button>`;
  const footer = htl.html`
    <div>
      <hr style="padding:0;margin:10px 0">
      <div style="display:flex;gap:1ch;justify-style: ${justify} ">
        ${submit}
      </div>
    </div>
  `;
  const template = inputs => {
    const row = document.createElement("div");
    row.className = "input-toolbar";
    row.append(countyView);
    (Array.isArray(inputs) ? inputs : Object.values(inputs))
      .forEach(input => row.append(input));
    const container = document.createElement("div");
    container.style.cssText = "width:100%;max-width:100%;min-width:0;box-sizing:border-box";
    container.append(row, footer);
    return container;
  };
  const input = fn({ submit, reset, footer, template, onSubmit, onReset });
  let invalid = !!input.querySelector(":invalid");
  submit.disabled = !resubmit || invalid;
  input.addEventListener("input", onInnerInputCapture, { capture: true });
  input.addEventListener("input", onInnerInput);
  let value = required ? undefined : input.value;
  const wrapper = htl.html`<div style="width: ${width} ">${input}`;
  wrapper.addEventListener("submit", onSubmit);
  return Object.defineProperty(wrapper, "value", {
    get: () => value,
    set: v => { input.value = v }
  });
}

// Define years to plot (2021-2015) and ELA3_8
gradeSequence = {
  if (!inputSels) return [];
  // no longer have a user selection for grade so hardcode it
  // const g = inputSels.grade;
  const g = "ELA3_8";
  return [0, 1, 2, 3, 4]
    .filter(offset => g === "ELA3_8" || (g - offset) >= 3)
    .map(offset => ({
      year: baseYear - offset,
      assessment: g === "ELA3_8" ? "ELA3_8" : `ELA${g - offset}`
    }));
}
shared_dataFiltered = {
  if (!inputSels) return [];
  const district = inputSels.district.trim().toUpperCase();
  return shared_data
    .filter(d => d.ENTITY_NAME.toUpperCase() === district)
    .sort((a, b) => Number(a.YEAR) - Number(b.YEAR));
}

// Preprocess accumulation dataset for Waterfall Chart
waterfallData = {
  let acc = 0;
  return shared_dataFiltered.map(d => {
    const annual = Number(d.Special_Education_Expenditures) || 0;
    const yStart = acc;
    acc += annual;
    return {
      ...d,
      YEAR: String(d.YEAR),
      annual,
      yStart,
      yEnd: acc
    };
  });
}

// 1. Calculated Cumulative SPED Spending (2021-YEAR)
cumulativeTotal = {
  if (!inputSels || shared_dataFiltered.length === 0) return 0;
  const years = [2021, 2022, 2023, 2024, 2025];
  let sum = 0;
  years.forEach(y => {
    const row = shared_dataFiltered.find(d => Number(d.YEAR) === y);
    if (row && row.Special_Education_Expenditures) {
      sum += +row.Special_Education_Expenditures;
    }
  });
  return sum;
}

// 2. Extract 2025 SPED spending per Pupil
spedPerPupil2025 = {
  if (!inputSels || shared_dataFiltered.length === 0) return 0;
  const row = shared_dataFiltered.find(d => Number(d.YEAR) === 2025);
  return (row && row.Special_Education_Expenditures_PerPupil) ? +row.Special_Education_Expenditures_PerPupil : 0;
}

Enrollment2025 = {
  if (!inputSels || shared_dataFiltered.length === 0) return 0;
  const row = shared_dataFiltered.find(d => Number(d.YEAR) === 2025);
  return (row && row.PK12_Enrollment) ? +row.PK12_Enrollment : 0;
}
shared_dataFiltered_ELA = {
  if (!inputSels) return [];
  const district_ela = inputSels.district.trim().toUpperCase();
  return shared_data
    .filter(d => d.ENTITY_NAME.toUpperCase() === district_ela &&
    gradeSequence.some(g => Number(g.year) === Number(d.YEAR) && g.assessment === d.ASSESSMENT_NAME))
    .sort((a, b) => Number(a.YEAR) - Number(b.YEAR));
}

// label for the assessment, used on the x-axis

  ? "All Grades 3-8"
  : `Grade ${assessment.replace("ELA", "")}`
// ELA plot.  Subgroup and assesment will never actually change being there's not user input for them
function subgroupRows(subgroupName) {

  const byYear = [...gradeSequence]
    .sort((a, b) => a.year - b.year)
    .map(({ year, assessment }) => {
      const row = shared_dataFiltered_ELA.find(d =>
        Number(d.YEAR) === Number(year) &&
        d.ASSESSMENT_NAME === assessment &&
        d.SUBGROUP_NAME === subgroupName
      );

      return {
        YEAR: year,
        label: `${year}\n${gradeLabel(assessment)}`,
        Prof_Count: row ? +row.Prof_Count : 0,
        NotProf_Count: row ? +row.NotProf_Count : 0,
        NOT_TESTED: row ? +row.NOT_TESTED : 0,
        Prof_Pct: row ? +row.Prof_Pct : null,
        NotProf_Pct: row ? +row.NotProf_Pct : null,
        PER_PROF: row ? +row.PER_PROF : null
      };
    });

  // console.log("subgroupRows - byYear:", byYear);

  const result = byYear.flatMap(d => [
    { label: d.label, YEAR: d.YEAR, category: "Proficient", count: d.Prof_Count, pct: d.Prof_Pct, pct2: d.PER_PROF },
    { label: d.label, YEAR: d.YEAR, category: "Not Proficient", count: d.NotProf_Count, pct: null },
    { label: d.label, YEAR: d.YEAR, category: "Not Tested", count: d.NOT_TESTED, pct: null }
  ]);

  // console.log("subgroupRows flatMap - result:", result);

  return result;
}
