// Guard function - toolbar architecture (county -> district submit/reset)
function guard(fn, options = {}, countyView) {
  const {
    submitLabel = "Submit",
    resetLabel = "Reset",
    required = false,
    resubmit = true,
    width = "100%",
    justify = "start"
  } = options;

  const onSubmit = () => {
    value = input.value;
    submit.disabled = !resubmit || invalid;
    reset.disabled = true;
    wrapper.dispatchEvent(new Event("input", { bubbles: true }));
  };
  const onReset = () => {
    input.value = value;
    submit.disabled = !resubmit || invalid;
    reset.disabled = true;
  };
  const onInnerInputCapture = () => {
    invalid = true;
    submit.disabled = true;
  };
  const onInnerInput = e => {
    e.stopPropagation();
    submit.disabled = false;
    reset.disabled = false;
  };
  const submit = htl.html`<button ${ { disabled: !resubmit && !required, onclick: onSubmit } } class="btn btn-primary btn-sm">${submitLabel}</button>`;
  const reset = htl.html`<button ${ { disabled: true, onclick: onReset } } class="btn btn-secondary btn-sm">${resetLabel}</button>`;
  const footer = htl.html`
    <div>
      <hr style="padding:0;margin:10px 0">
      <div style="display:flex;gap:1ch;justify-style: ${justify} ">
        ${submit}
      </div>
    </div>
  `;
  const template = inputs => {
    const row = document.createElement("div");
    row.className = "input-toolbar";
    row.append(countyView);
    (Array.isArray(inputs) ? inputs : Object.values(inputs))
      .forEach(input => row.append(input));
    const container = document.createElement("div");
    container.style.cssText = "width:100%;max-width:100%;min-width:0;box-sizing:border-box";
    container.append(row, footer);
    return container;
  };
  const input = fn({ submit, reset, footer, template, onSubmit, onReset });
  let invalid = !!input.querySelector(":invalid");
  submit.disabled = !resubmit || invalid;
  input.addEventListener("input", onInnerInputCapture, { capture: true });
  input.addEventListener("input", onInnerInput);
  let value = required ? undefined : input.value;
  const wrapper = htl.html`<div style="width: ${width} ">${input}`;
  wrapper.addEventListener("submit", onSubmit);
  return Object.defineProperty(wrapper, "value", {
    get: () => value,
    set: v => { input.value = v }
  });
}

spendingColor = {
  // Shared color palette across all charts
  const EquipMaterialShades = d3.quantize(t => d3.interpolateYlOrBr(0.1 + t * 0.2), 3);
  const SPEDContracts  = d3.quantize(t => d3.interpolateGreens(.5), 1); 
  const legalShades  = d3.quantize(t => d3.interpolateReds(0.5 + t * 0.7), 3); 
  
  // Shades of Purple for Psychological, Social Work, and Special Schools (A2820_, A2825_, A2830_)
  const psychShades   = d3.quantize(t => d3.interpolatePurples(0.25 + t * 0.25), 3); // Lighter Purples
  const socialShades  = d3.quantize(t => d3.interpolatePurples(0.55 + t * 0.25), 3); // Mid Purples
  const specSchShades = d3.quantize(t => d3.interpolatePurples(0.82 + t * 0.15), 3); // Dark Purples

  // Shared shades of blue across all 3 plots for tuition and BOCES services
  const tuitionBlue1 = d3.interpolateBlues(0.30); // Tuition To Public Schools
  const tuitionBlue2 = d3.interpolateBlues(0.50); // Tuition To Other
  const tuitionBlue3 = d3.interpolateBlues(0.60); // Tuition To Charter Schools
  const tuitionBlue4 = d3.interpolateBlues(0.80); // BOCES Services (Lighter Blue)

  const colorMap = new Map([
    ["Legal - Contractual & Other", legalShades[0]],
    ["Legal - Materials & Supplies", legalShades[1]],
    ["Legal - BOCES Services", legalShades[2]],

    ["Psychological - Contractual & Other", psychShades[0]],
    ["Psychological - Materials & Supplies", psychShades[1]],
    ["Psychological - BOCES Services", psychShades[2]],

    ["Social Work - Contractual & Other", socialShades[0]],
    ["Social Work - Materials & Supplies", socialShades[1]],
    ["Social Work - BOCES Services", socialShades[2]],

    ["Special Schools - Contractual & Other", specSchShades[0]],
    ["Special Schools - Materials & Supplies", specSchShades[1]],
    ["Special Schools - BOCES Services", specSchShades[2]],

    ["SPED - Equipment", EquipMaterialShades[0]],
    ["SPED - Materials & Supplies", EquipMaterialShades[1]],
    ["SPED - Textbooks", EquipMaterialShades[2]],

    ["SPED - Contractual & Other", SPEDContracts[0]],

    ["SPED - Tuition To Public Schools", tuitionBlue1],
    ["SPED - Tuition To Other", tuitionBlue2],
    ["SPED - Tuition To Charter Schools", tuitionBlue3],
    ["SPED - BOCES Services", tuitionBlue4],
    ["SPED - Transportation", d3.interpolateBlues(0.95)]
  ]);

  return d3.scaleOrdinal()
    .domain(spedFields.map(f => f.label))
    .range(spedFields.map(f => {
      // Normalize single vs double space if present
      const normName = f.item_name.replace(/^SPED\s+-\s+/, "SPED - ");
      if (colorMap.has(normName)) return colorMap.get(normName);
      if (colorMap.has(f.item_name)) return colorMap.get(f.item_name);

    }));
}
// Color scale for the Special Funds

sfColor = {
  // Groups with a single static color (no gradient) go here.
  const staticColors = {
    "Tuition To Public Schools":        d3.interpolateBlues(0.30),
    "Tuition To Other":                 d3.interpolateBlues(0.50),
    "Tuition To Charter Schools":       d3.interpolateBlues(0.60),
    "BOCES Services":                  d3.interpolateBlues(0.80),
    "Transportation":                  d3.interpolateBlues(0.90),
    "Transportation Contract":         d3.interpolateBlues(0.92),
    "Transportation Public":           d3.interpolateBlues(0.94),
    "Transportation BOCES":            d3.interpolateBlues(0.97),
    "Legal - Contractual & Other":     d3.interpolateReds(.2),
    "Legal - Materials & Supplies":    d3.interpolateReds(.3),
    "Legal - BOCES Services":          d3.interpolateReds(.35),
    "Contractual & Other":             d3.interpolateGreens(0.5)                     
  };
  // Groups shaded across a range of a d3 interpolator go here.
  const interps = {
    "Equipment":            d3.interpolateYlOrBr,
    "Materials & Supplies": d3.interpolateYlOrBr,
    "Textbooks":            d3.interpolateYlOrBr
  };
  // Per-color lightness band [min, max]; anything not listed uses the default.
  const defaultRange = [0.4, 0.9];
  const ranges = {
    "Equipment":            [0.1, 0.35],
    "Materials & Supplies": [0.1, 0.35],
    "Textbooks":            [0.1, 0.35]
  };
  const groups = d3.group(sfFields, f => f.item_name);
  const colorOf = new Map();
  for (const [g, members] of groups) {
    if (staticColors[g]) {
      members.forEach(f => colorOf.set(f.label, staticColors[g]));
      continue;
    }
    const interp = interps[g] || d3.interpolateGreys;
    const [lo, hi] = ranges[g] || defaultRange;
    members.forEach((f, i) => {
      const t = members.length === 1 ? 0.7 : i / (members.length - 1);
      colorOf.set(f.label, interp(lo + (hi - lo) * t));
    });
  }
  return d3.scaleOrdinal()
    .domain(sfFields.map(f => f.label))
    .range(sfFields.map(f => colorOf.get(f.label)));
}

itemColor = {
  const byItem = new Map();
  for (const f of sfFields) {
    let combinedLabel = "SPED - " + f.item_name;
    if (f.item_name.startsWith("Transportation")) combinedLabel = "SPED - Transportation";
    byItem.set(combinedLabel, sfColor(f.label));
  }
  for (const f of spedFields) {
    if (f.item_name.startsWith("Legal")) {
      byItem.set("Legal", d3.interpolateReds(0.5));
    } else if (f.item_name.startsWith("Psychological")) {
      byItem.set("Psychological", d3.interpolatePurples(0.35));
    } else if (f.item_name.startsWith("Social Work")) {
      byItem.set("Social Work", d3.interpolatePurples(0.65));
    } else if (f.item_name.startsWith("Special Schools")) {
      byItem.set("Special Schools", d3.interpolatePurples(0.88));
    } else if (f.item_name.startsWith("Transportation")) {
      byItem.set("SPED - Transportation", d3.interpolateBlues(0.92));
    } else {
      const normName = f.item_name.replace(/^SPED\s+-\s+/, "SPED - ");
      if (!byItem.has(normName)) {
        byItem.set(normName, spendingColor(f.label));
      }
    }
  }

  const masterOrder = [
    "Legal",
    "Psychological",
    "Social Work",
    "Special Schools",
    "SPED - Equipment",
    "SPED - Materials & Supplies",
    "SPED - Textbooks",
    "SPED - Contractual & Other",
    "SPED - Tuition To Public Schools",
    "SPED - Tuition To Other",
    "SPED - Tuition To Charter Schools",
    "SPED - BOCES Services",
    "SPED - Transportation"
  ];

  const sortedKeys = [...byItem.keys()].sort((a, b) => {
    let idxA = masterOrder.indexOf(a);
    let idxB = masterOrder.indexOf(b);
    if (idxA === -1) idxA = 999;
    if (idxB === -1) idxB = 999;
    return idxA - idxB;
  });

  return d3.scaleOrdinal()
    .domain(sortedKeys)
    .range(sortedKeys.map(k => byItem.get(k)));
}

combinedChartData = {
  const totals = new Map();
  function addFields(fields) {
    for (const f of fields) {
      const v = +selectedRow?.[f.key] || 0;
      let labelName = f.item_name;
      if (f.item_name.startsWith("Legal")) {
        labelName = "Legal";
      } else if (f.item_name.startsWith("Psychological")) {
        labelName = "Psychological";
      } else if (f.item_name.startsWith("Social Work")) {
        labelName = "Social Work";
      } else if (f.item_name.startsWith("Special Schools")) {
        labelName = "Special Schools";
      } else if (f.item_name.startsWith("Transportation")) {
        labelName = "SPED - Transportation";
      } else {
        const cleanName = f.item_name.replace(/^SPED\s+-\s+/, "");
        labelName = "SPED - " + cleanName;
      }
      
      totals.set(labelName, (totals.get(labelName) || 0) + v);
    }
  }
  if (selectedRow) {
    addFields(spedFields);
    addFields(sfFields);
  }

  const masterOrder = [
    "Legal",
    "Psychological",
    "Social Work",
    "Special Schools",
    "SPED - Equipment",
    "SPED - Materials & Supplies",
    "SPED - Textbooks",
    "SPED - Contractual & Other",
    "SPED - Tuition To Public Schools",
    "SPED - Tuition To Other",
    "SPED - Tuition To Charter Schools",
    "SPED - BOCES Services",
    "SPED - Transportation"
  ];

  return Array.from(totals, ([label, value]) => ({ label, value }))
    .filter(d => d.value > 0)
    .sort((a, b) => {
      let idxA = masterOrder.indexOf(a.label);
      let idxB = masterOrder.indexOf(b.label);
      if (idxA === -1) idxA = 999;
      if (idxB === -1) idxB = 999;
      return idxA - idxB;
    });
}

// Reusable pie chart (d3), with a % label on larger slices and a legend.
function pieChart(data, { title = "", color = spendingColor, maxHeight = 350 } = {}) {
  if (!data || data.length === 0) {
    return html`<div style="text-align:center;color:#888;padding:20px;">
      <strong>${title}</strong><br>No spending reported for this district.
    </div>`;
  }

  const total = d3.sum(data, d => d.value);
  const radius = 180;
  const padTop = title ? 30 : 0;
  const drawWidth = (radius + 10) * 2;
  const drawHeight = (radius + 10) * 2 + padTop;
  const maxWidth = maxHeight * (drawWidth / drawHeight);

  const arcs = d3.pie().value(d => d.value).sort(null)(data);
  const arcGen = d3.arc().innerRadius(radius * 0.35).outerRadius(radius);

  const svg = d3.create("svg")
    .attr("viewBox", [-drawWidth / 2, -drawHeight / 2 - padTop / 2, drawWidth, drawHeight])
    .style("width", "100%")
    .style("height", "auto")
    .style("max-width", `${maxWidth}px`)
    .style("display", "block")
    .style("margin", "0 auto")
    .style("font", "12px sans-serif");

  svg.append("g")
    .selectAll("path")
    .data(arcs)
    .join("path")
      .attr("d", arcGen)
      .attr("fill", d => color(d.data.label))
      .attr("stroke", "white")
      .attr("stroke-width", 1)
    .append("title")
      .text(d => `${d.data.label}
$${Math.round(d.data.value).toLocaleString()} (${((d.data.value / total) * 100).toFixed(1)}%)`);

  svg.append("g")
    .attr("text-anchor", "middle")
    .selectAll("text")
    .data(arcs)
    .join("text")
      .attr("transform", d => `translate(${arcGen.centroid(d)})`)
      .style("font-size", "10px")
      .style("font-weight", "600")
      .style("fill", "#fff")
      .text(d => (d.data.value / total) > 0.04 ? `${Math.round((d.data.value / total) * 100)}%` : "");

  if (title) {
    svg.append("text")
        .attr("x", 0)
        .attr("y", -drawHeight / 2 + 16)
        .attr("text-anchor", "middle")
        .style("font-size", "15px")
        .style("font-weight", "bold")
        .text(title);
  }

const legend = html`<div style="font-size:11px;margin:6px auto 0;width:fit-content;max-width:100%;">
  ${data.map(d => html`<div style="display:flex;align-items:center;justify-content:flex-start;gap:6px;margin:2px 0;">
    <span style="width:10px;height:10px;background:${color(d.label)};display:inline-block;border-radius:2px;flex:none;"></span>
    <span>${d.label}: <strong>$${Math.round(d.value).toLocaleString()}</strong></span>
  </div>`)}
</div>`;

return html`<div style="text-align:left;width:100%;margin-top:0;">${svg.node()}${legend}</div>`;
}
