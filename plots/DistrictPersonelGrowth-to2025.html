// guard function
// makes a submit and reset button to use after multiple inputs made, see https://observablehq.com/@mootari/inputs-submit

function guard(fn, options = {}, countyView) {
  const {
    submitLabel = "Submit",
    resetLabel = "Reset",
    required = false,
    resubmit = true,
    width = "100%",
    justify = "start"
  } = options;

  const onSubmit = () => {
    value = input.value;
    submit.disabled = !resubmit || invalid;
    reset.disabled = true;
    wrapper.dispatchEvent(new Event("input", { bubbles: true }));
  };

  const onReset = () => {
    input.value = value;
    submit.disabled = !resubmit || invalid;
    reset.disabled = true;
  };

  const onInnerInputCapture = () => {
    invalid = true;
    submit.disabled = true;
  };

  const onInnerInput = e => {
    e.stopPropagation();
    submit.disabled = false;
    reset.disabled = false;
  };

  const submit = htl.html`
    <button ${{disabled: !resubmit && !required, onclick: onSubmit}}>
      ${submitLabel}
  `;

  const reset = htl.html`
    <button ${{disabled: true, onclick: onReset}}>
      ${resetLabel}
  `;

  const footer = htl.html`
    <div>
      <hr style="padding:0;margin:10px 0">
      <div style="display:flex;gap:1ch;justify-content:${justify}">
        ${submit}
      </div>
    </div>
  `;

const template = inputs => {
  const row = document.createElement("div");
  row.className = "input-toolbar";

  row.append(countyView);

  (Array.isArray(inputs) ? inputs : Object.values(inputs))
    .forEach(input => row.append(input));

  const container = document.createElement("div");
  container.style.cssText =
    "width:100%;max-width:100%;min-width:0;box-sizing:border-box";

  container.append(row, footer);

  return container;
};

  const input = fn({
    submit,
    reset,
    footer,
    template,
    onSubmit,
    onReset
  });

  let invalid = !!input.querySelector(":invalid");
  submit.disabled = !resubmit || invalid;

  input.addEventListener("input", onInnerInputCapture, {capture: true});
  input.addEventListener("input", onInnerInput);

  let value = required ? undefined : input.value;

  const wrapper = htl.html`<div style="width:${width}">${input}`;

  wrapper.addEventListener("submit", onSubmit);

  return Object.defineProperty(wrapper, "value", {
    get: () => value,
    set: v => { input.value = v }
  });
}

function changePlot(data, width, height) {

  return Plot.plot({
    className: "StudentStaffgrowth-plot",

    width,
    height,
    marginLeft: 80,
    marginBottom: 50,
    marginRight: 30,

    x: { label: "School year (ending)", tickFormat: "d", ticks: yearTicks, labelAnchor: "center", labelArrow: false },
    y: { label: `Change since ${firstYear}`, grid: true, labelAnchor: "center", ticks: 7, labelArrow: false},
    color: {
      domain: [
        "Students",
        "Classroom Teachers",
        "Aministrators, Priciples, Other non-Inst.Staff"
      ],
      range: ["#4f81a8", "#E08A1E", "#d95c5c"],
      legend: true
    },
    marks: [
      Plot.ruleY([0]),
      Plot.line(data, {x: "YEAR", y: "change", stroke: "series", strokeWidth: 2.5}),
      Plot.dot(data, {x: "YEAR", y: "change", stroke: "series", r: 4}),
      Plot.tip(data, Plot.pointer({
        x: "YEAR", y: "change", stroke: "series",
        title: d => `${d.series}\n${d.YEAR}\nChange: ${d.change.toFixed(1)}\nTotal: ${d.value.toFixed(1)}`
      }))
    ]
  });
}
